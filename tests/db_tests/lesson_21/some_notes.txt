from unicodedata import category

CREATE TABLE public.dogs (
	id serial4 NOT NULL,
	name varchar NULL,
	breed varchar NULL,
	CONSTRAINT dogs_pk PRIMARY KEY (id)
);

CREATE TABLE public.products (
	id serial4 NOT NULL,
	"name" varchar NOT NULL,
	description varchar NULL,
	price int DEFAULT 1 NOT NULL,
	CONSTRAINT products_pk PRIMARY KEY (id),
	CONSTRAINT products_unique UNIQUE ("name"),
	category int NOT NULL
);

CREATE TABLE public.categories (
	category_id serial4 NOT NULL,
	category_name varchar NOT NULL,
	storage_condition varchar NOT NULL,
	CONSTRAINT categories_pk PRIMARY KEY (category_id)
);

ALTER TABLE public.products ADD CONSTRAINT products_categories_fk FOREIGN KEY (category) REFERENCES public.categories(category_id);

select "name", category_name from public.products join public.categories on public.products.category = public.categories.category_id
